```typescript
// src/lib/scorecard/types.ts
export type DimensionKey =
  | "routing" | "revenue" | "liquidity" | "channelHealth"
  | "forceClose" | "restart" | "peer"
  | "stability" | "pgHealth" | "visibility";
export const COMPUTABLE_DIMENSIONS: readonly DimensionKey[]; // first 7
export const NO_DATA_DIMENSIONS: readonly DimensionKey[];    // stability, pgHealth, visibility (D-01)
export type Letter = "A" | "B" | "C" | "D" | "F";
export type StatusColor = "ok" | "warn" | "critical" | "nodata";
export interface DimensionResult {
  raw: number | null;          // RAW units (sats, ratio, count, PPM) — formatter at render only (AFM-002)
  asOf: Date | null;
  note?: string;
  extras?: Record<string, number>; // e.g. maxHourlySelfCloses, earnings7dSat, rebalanceSuccessRate
}
export interface Tier1Breach { label: string }   // e.g. "fee crash"
export interface GradedDimension {
  key: DimensionKey;
  raw: number | null;
  pct: number | null;          // null for no-data dims
  letter: Letter | null;
  color: StatusColor;
  tier1Breach: Tier1Breach | null;
  note?: string;
  asOf: Date | null;
}
export interface PeerRelative {
  suppressed: boolean;         // true → "no peer baseline yet" (D-11)
  median?: number;
  deltaFromMedian?: number;
  isOutlier?: boolean;         // D-12 — NEVER alters grade
}
export interface ScorecardView {
  pubkey: string;
  alias: string | null;
  backend: "lnd" | "ldk";
  compositePct: number | null;
  compositeLetter: Letter | null;
  dot: "ok" | "warn" | "critical";   // D-07: critical iff ≥1 Tier-1 breach
  reason: string | null;             // D-06 reason subtext
  criticalChips: string[];           // tier1 labels, rendered as "critical: {label}"
  dimensions: Array<GradedDimension & { peerRelative: PeerRelative | null }>;
  asOf: Date | null;                 // D-13 worst-case stamp
  stale: boolean;
}
export interface FleetView { nodes: ScorecardView[]; fetchedAt: Date }

// src/lib/scorecard/config.ts
export type Direction = "higher-better" | "lower-better" | "mid-better";
export interface Band { a: number; b: number; c: number; d: number; direction: Direction; center?: number }
export interface Tier1Line { metric: "raw" | string; op: ">" | "<" | ">=" | "<="; value: number; label: string }
export interface DimConfig { band: Band; tier1?: Tier1Line; reasonLabel: string }
export type BackendConfig = Record<Exclude<DimensionKey, "stability" | "pgHealth" | "visibility">, DimConfig>;
export const SCORECARD_CONFIG: Record<"lnd" | "ldk", BackendConfig>;
export const COMPARE_CONFIG: { madN: number; minCohortForMedian: number };   // { 3, 3 }
export const STALENESS: { warnAfterHours: number };                          // { 6 }

// src/lib/scorecard/grade.ts
export function pctFor(raw: number, band: Band): number;                       // 0..100
export function letterFor(pct: number): Letter;                                // A≥90 B≥80 C≥70 D≥60 else F
export function gradeDimension(key: DimensionKey, result: DimensionResult, cfg: DimConfig): GradedDimension;
export function gradeNoData(key: DimensionKey): GradedDimension;               // pct/letter null, color "nodata"
export function compositeOf(dims: GradedDimension[]): { pct: number; letter: Letter } | null;
export function nodeDot(dims: GradedDimension[]): "ok" | "warn" | "critical";
export function reasonFor(dims: GradedDimension[], cfg: BackendConfig): string | null;
export function criticalChips(dims: GradedDimension[]): string[];

// src/lib/scorecard/compare.ts
export function median(xs: number[]): number;
export function mad(xs: number[]): number;
export function outlierCheck(value: number, cohort: number[], madN?: number, minCohort?: number):
  { suppressed: true } | { suppressed: false; median: number; isOutlier: boolean; deltaFromMedian: number };
export function outlierFor(value: number, cohort: number[], direction: Direction, madN?: number, minCohort?: number): PeerRelative;

// src/lib/scorecard/freshness.ts
export function worstCaseAsOf(asOfs: Array<Date | null>): Date | null;  // MIN of non-null, null if none
export function isStale(asOf: Date | null, warnAfterHours: number, now?: Date): boolean; // null asOf → true

// src/lib/config/backends.ts
export function backendFor(pubkey: string): Backend;  // env NODE_BACKENDS JSON override → DEFAULT_BACKENDS → "lnd"
```
