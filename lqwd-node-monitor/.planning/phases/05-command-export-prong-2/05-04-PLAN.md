// ExportBlock { title, contextLines: {key,value}[], safetyLines: string[],
//               steps: { label, command: string|null, note: string|null }[], copyAll: string|null }

// Copy handler pattern (RESEARCH §Code Examples — use verbatim):
async function copy(text: string, setState: (s: "copied" | "failed") => void) {
  try {
    if (!navigator.clipboard) throw new Error("clipboard unavailable");
    await navigator.clipboard.writeText(text);
    setState("copied");
  } catch {
    setState("failed"); // locked honest failure copy; never silent
  }
}
